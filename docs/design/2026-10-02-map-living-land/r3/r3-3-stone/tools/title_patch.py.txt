#!/usr/bin/env python3
"""Lane R3.2 scratch: arm a measuring worktree (already armed by
tools/map_trace/qa_patch.py) with the title probe: copy it in as
tools/qa_title_probe.gd, stamp its build, and add --title-trace-probe=<mode>
to that worktree's main.gd (uncommitted, never pushed)."""
import subprocess, sys
from pathlib import Path
HERE = Path(__file__).resolve().parent
wt = Path(sys.argv[1])
main = wt / "application/main.gd"
text = main.read_text()
if "--title-trace-probe" not in text:
    edits = [
        ('''		elif arg == "--launch-timing":
			launch_timing = true''', '''		elif arg == "--launch-timing":
			launch_timing = true
		elif arg.begins_with("--title-trace-probe="):
			title_trace_probe = true'''),
        ('''	var launch_timing: bool = false''', '''	var launch_timing: bool = false
	var title_trace_probe: bool = false'''),
        ('''	if resume_run:
		_continue_run(_load_run())''', '''	if title_trace_probe:
		var title_probe: Variant = (load("res://tools/qa_title_probe.gd") as GDScript).new()
		if title_probe is Node:
			var title_node: Node = title_probe
			add_child(title_node)
	if resume_run:
		_continue_run(_load_run())'''),
    ]
    for a, b in edits:
        assert text.count(a) == 1, a
        text = text.replace(a, b)
    main.write_text(text)
head = subprocess.run(["git", "-C", str(wt), "rev-parse", "--short=8", "HEAD"], capture_output=True, text=True, check=True).stdout.strip()
probe = (HERE / "title_probe.gd").read_text()
assert probe.count('const BUILD: String = "unstamped"') == 1
(wt / "tools/qa_title_probe.gd").write_text(probe.replace('const BUILD: String = "unstamped"', 'const BUILD: String = "%s"' % head))
print("title probe armed:", wt, head)
