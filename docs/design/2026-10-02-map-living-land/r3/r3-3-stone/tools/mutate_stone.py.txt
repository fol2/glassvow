only = sys.argv[3].split(",") if len(sys.argv) > 3 else None
results = []
for i, (name, rel, before, after) in enumerate(M):
    if only and str(i) not in only:
        continue
    path = WT / rel
    backup = OUT / ("backup-%d" % i)
    shutil.copy2(path, backup)
    try:
        text = path.read_text()
        if text.count(before) != 1:
            results.append((name, "ANCHOR NOT FOUND"))
            print("%-44s %s" % results[-1], flush=True)
            continue
        path.write_text(text.replace(before, after))
        run = subprocess.run(["godot", "--headless", "-s", "res://tests/run_all.gd", "--", "--tests=res://tests/test_map_stone.gd"],
                             cwd=WT, capture_output=True, text=True, timeout=1500)
        out = run.stdout + run.stderr
        failed = "FAIL res://tests/test_map_stone.gd" in out or "PASS (1 tests)" not in out
        lines = [l for l in out.splitlines() if l.startswith("  - test_map_stone")]
        results.append((name, ("caught: " + "; ".join(l.strip()[18:] for l in lines[:4])) if failed and lines
                        else ("caught (no assertion line; parse or run error)" if failed else "SURVIVED")))
    finally:
        shutil.copy2(backup, path)
    print("%-44s %s" % results[-1], flush=True)
print("caught %d of %d" % (sum(1 for r in results if r[1].startswith("caught")), len(results)))
